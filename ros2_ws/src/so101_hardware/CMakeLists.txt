cmake_minimum_required(VERSION 3.16)
project(so101_hardware)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
find_package(hardware_interface REQUIRED)
find_package(pluginlib REQUIRED)
find_package(rclcpp REQUIRED)
find_package(realtime_tools REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(std_msgs REQUIRED)
find_package(trajectory_msgs REQUIRED)
add_library(so101_topic_system SHARED src/topic_system.cpp)
target_compile_features(so101_topic_system PUBLIC cxx_std_17)
ament_target_dependencies(so101_topic_system hardware_interface pluginlib rclcpp realtime_tools sensor_msgs std_msgs trajectory_msgs)
pluginlib_export_plugin_description_file(hardware_interface plugins.xml)
install(TARGETS so101_topic_system DESTINATION lib)
ament_python_install_package(${PROJECT_NAME})
install(PROGRAMS scripts/so101_leader_driver scripts/so101_follower_driver scripts/camera_capture DESTINATION lib/${PROJECT_NAME})
install(FILES LICENSE DESTINATION share/${PROJECT_NAME})
ament_package()
